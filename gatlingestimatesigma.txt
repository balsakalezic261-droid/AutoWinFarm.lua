local function SetToggleVisual(enabled)
    if enabled then
        TweenService:Create(
            Toggle,
            TweenInfo.new(0.18),
            {BackgroundColor3 = Color3.fromRGB(105, 65, 165)}
        ):Play()

        TweenService:Create(
            Knob,
            TweenInfo.new(0.18),
            {Position = UDim2.new(1, -23, 0, 3)}
        ):Play()

        Status.Text = "Auto DJ: ON"
        Status.TextColor3 = Color3.fromRGB(190, 160, 245)
    else
        TweenService:Create(
            Toggle,
            TweenInfo.new(0.18),
            {BackgroundColor3 = Color3.fromRGB(55, 55, 67)}
        ):Play()

        TweenService:Create(
            Knob,
            TweenInfo.new(0.18),
            {Position = UDim2.fromOffset(3, 3)}
        ):Play()

        Status.Text = "Auto DJ: OFF"
        Status.TextColor3 = Color3.fromRGB(125, 125, 140)
    end
end

local function SetAutoDJ(enabled)

    getgenv().AutoDJ = enabled

    if enabled then

        if AutoDJ then
            return
        end

        SetToggleVisual(true)

        AutoDJ = task.spawn(function()

            local success, err = pcall(function()
                RunAutoDJ()
            end)

            AutoDJ = nil

            if not success then
                warn("[TDS Hub] Auto DJ error:", err)
            end

            if getgenv().AutoDJ then
                getgenv().AutoDJ = false
                SetToggleVisual(false)
            end
        end)

    else

        SetToggleVisual(false)

        if AutoDJ then
            task.cancel(AutoDJ)
            AutoDJ = nil
        end
    end
end

local UserInputService = game:GetService("UserInputService")

local dragging = false
local dragStart
local startPosition
local dragInput

local function updateDrag(input)
    local delta = input.Position - dragStart

    Window.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end

Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Window.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then

        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        updateDrag(input)
    end
end)
